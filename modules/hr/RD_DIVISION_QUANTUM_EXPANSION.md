# modules/hr/rd_productization.py

from dataclasses import dataclass
from datetime import datetime
from enum import Enum
from typing import List, Dict, Optional
import numpy as np

class ProjectStage(Enum):
    """Research-to-product pipeline stages"""
    RESEARCH = "research"                    # Pure research phase
    PROOF_OF_CONCEPT = "proof_of_concept"   # POC validation
    PROTOTYPE = "prototype"                  # Working prototype
    ALPHA = "alpha"                          # Alpha testing
    BETA = "beta"                            # Beta release
    PRODUCTION = "production"                # Production ready
    MAINTENANCE = "maintenance"              # Ongoing support

class ProjectType(Enum):
    """Types of R&D projects"""
    MICROSERVICE = "microservice"           # Standalone service
    MODULE = "module"                        # Integrated module
    TOOL = "tool"                            # CLI/utility tool
    ALGORITHM = "algorithm"                  # Algorithm library
    FRAMEWORK = "framework"                  # Development framework
    RESEARCH_PAPER = "research_paper"       # Publication

@dataclass
class ResearchProject:
    """Research project with productization tracking"""
    
    project_id: str
    name: str
    project_type: ProjectType
    stage: ProjectStage
    
    # Team
    lead_researcher: str                     # Primary researcher
    team_members: List[str]                  # Collaborating team
    
    # Progress tracking
    start_date: datetime
    target_completion: datetime
    current_milestone: str
    completion_percentage: float             # 0.0 - 1.0
    
    # Technical details
    repository_path: Optional[str]           # Git repo location
    documentation_path: Optional[str]        # Docs location
    key_technologies: List[str]              # Tech stack
    
    # Productization
    production_readiness: float              # 0.0 - 1.0
    deployment_complexity: str               # "low", "medium", "high"
    maintenance_requirements: str            # Support needs
    
    # Success metrics
    research_impact: float                   # Research value (0.0 - 1.0)
    product_viability: float                 # Commercial viability (0.0 - 1.0)
    innovation_score: float                  # Novelty rating (0.0 - 1.0)
    
    # Quantum attributes
    team_coherence: float                    # Team synergy (0.0 - 1.0)
    project_entanglements: Dict[str, float]  # Cross-project dependencies


class RDProductizationPipeline:
    """Research-to-product pipeline management"""
    
    def __init__(self):
        self.active_projects: Dict[str, ResearchProject] = {}
        self.completed_projects: List[ResearchProject] = []
        
    def create_project(
        self,
        project_id: str,
        name: str,
        project_type: ProjectType,
        lead_researcher: str,
        team_members: List[str],
        key_technologies: List[str]
    ) -> ResearchProject:
        """Initialize new R&D project"""
        
        project = ResearchProject(
            project_id=project_id,
            name=name,
            project_type=project_type,
            stage=ProjectStage.RESEARCH,
            lead_researcher=lead_researcher,
            team_members=team_members,
            start_date=datetime.now(),
            target_completion=None,
            current_milestone="Initial research phase",
            completion_percentage=0.0,
            repository_path=None,
            documentation_path=None,
            key_technologies=key_technologies,
            production_readiness=0.0,
            deployment_complexity="unknown",
            maintenance_requirements="TBD",
            research_impact=0.0,
            product_viability=0.0,
            innovation_score=0.0,
            team_coherence=0.0,
            project_entanglements={}
        )
        
        self.active_projects[project_id] = project
        return project
    
    def advance_stage(
        self,
        project_id: str,
        new_stage: ProjectStage,
        milestone: str
    ) -> ResearchProject:
        """Move project to next stage in pipeline"""
        
        project = self.active_projects[project_id]
        project.stage = new_stage
        project.current_milestone = milestone
        
        # Update completion percentage based on stage
        stage_progress = {
            ProjectStage.RESEARCH: 0.15,
            ProjectStage.PROOF_OF_CONCEPT: 0.30,
            ProjectStage.PROTOTYPE: 0.50,
            ProjectStage.ALPHA: 0.70,
            ProjectStage.BETA: 0.85,
            ProjectStage.PRODUCTION: 1.0,
            ProjectStage.MAINTENANCE: 1.0
        }
        
        project.completion_percentage = stage_progress[new_stage]
        
        return project
    
    def calculate_production_readiness(
        self,
        project_id: str,
        code_quality: float,          # 0.0 - 1.0
        documentation: float,          # 0.0 - 1.0
        test_coverage: float,          # 0.0 - 1.0
        performance: float,            # 0.0 - 1.0
        security: float                # 0.0 - 1.0
    ) -> float:
        """Calculate production readiness score"""
        
        # Weighted average of readiness factors
        weights = {
            'code_quality': 0.25,
            'documentation': 0.20,
            'test_coverage': 0.20,
            'performance': 0.20,
            'security': 0.15
        }
        
        readiness = (
            code_quality * weights['code_quality'] +
            documentation * weights['documentation'] +
            test_coverage * weights['test_coverage'] +
            performance * weights['performance'] +
            security * weights['security']
        )
        
        project = self.active_projects[project_id]
        project.production_readiness = readiness
        
        return readiness
    
    def calculate_team_coherence(
        self,
        project_id: str,
        team_member_profiles: Dict[str, np.ndarray]  # VSA vectors
    ) -> float:
        """Calculate quantum coherence for project team"""
        
        project = self.active_projects[project_id]
        team = [project.lead_researcher] + project.team_members
        
        # Get VSA vectors for team members
        vectors = [team_member_profiles.get(member) for member in team]
        vectors = [v for v in vectors if v is not None]
        
        if len(vectors) < 2:
            return 0.0
        
        # Calculate pairwise coherence
        coherences = []
        for i in range(len(vectors)):
            for j in range(i + 1, len(vectors)):
                similarity = np.dot(vectors[i], vectors[j])
                coherences.append(similarity)
        
        # Average coherence
        team_coherence = float(np.mean(coherences))
        project.team_coherence = team_coherence
        
        return team_coherence
    
    def get_parallel_project_capacity(
        self,
        team_member: str
    ) -> Dict[str, any]:
        """Check team member's parallel project capacity"""
        
        # Find all projects this member is involved in
        member_projects = [
            p for p in self.active_projects.values()
            if team_member in ([p.lead_researcher] + p.team_members)
        ]
        
        # Calculate workload
        current_load = sum(
            1.0 if p.lead_researcher == team_member else 0.3
            for p in member_projects
        )
        
        # Capacity thresholds
        max_capacity = 2.5  # Lead 2 projects + support 1-2 others
        
        return {
            'team_member': team_member,
            'active_projects': len(member_projects),
            'current_load': current_load,
            'max_capacity': max_capacity,
            'available_capacity': max(0, max_capacity - current_load),
            'projects': [p.project_id for p in member_projects]
        }
    
    def generate_pipeline_report(self) -> Dict[str, any]:
        """Generate comprehensive pipeline status report"""
        
        # Group projects by stage
        by_stage = {}
        for stage in ProjectStage:
            by_stage[stage.value] = [
                p for p in self.active_projects.values()
                if p.stage == stage
            ]
        
        # Calculate aggregate metrics
        total_projects = len(self.active_projects)
        avg_completion = np.mean([
            p.completion_percentage
            for p in self.active_projects.values()
        ]) if total_projects > 0 else 0.0
        
        avg_readiness = np.mean([
            p.production_readiness
            for p in self.active_projects.values()
        ]) if total_projects > 0 else 0.0
        
        avg_coherence = np.mean([
            p.team_coherence
            for p in self.active_projects.values()
            if p.team_coherence > 0
        ]) if total_projects > 0 else 0.0
        
        return {
            'timestamp': datetime.now().isoformat(),
            'total_active_projects': total_projects,
            'completed_projects': len(self.completed_projects),
            'projects_by_stage': {
                stage: len(projects)
                for stage, projects in by_stage.items()
            },
            'aggregate_metrics': {
                'average_completion': float(avg_completion),
                'average_production_readiness': float(avg_readiness),
                'average_team_coherence': float(avg_coherence)
            },
            'pipeline_health': self._calculate_pipeline_health(),
            'bottlenecks': self._identify_bottlenecks()
        }
    
    def _calculate_pipeline_health(self) -> str:
        """Assess overall pipeline health"""
        
        if not self.active_projects:
            return "idle"
        
        # Check for stalled projects
        stalled = sum(
            1 for p in self.active_projects.values()
            if p.completion_percentage < 0.2 and
            (datetime.now() - p.start_date).days > 90
        )
        
        # Check team coherence
        low_coherence = sum(
            1 for p in self.active_projects.values()
            if p.team_coherence < 0.60
        )
        
        total = len(self.active_projects)
        
        if stalled > total * 0.3:
            return "critical"
        elif low_coherence > total * 0.4:
            return "degraded"
        elif stalled > 0 or low_coherence > 0:
            return "warning"
        else:
            return "healthy"
    
    def _identify_bottlenecks(self) -> List[Dict[str, str]]:
        """Identify pipeline bottlenecks"""
        
        bottlenecks = []
        
        # Check for projects stuck in stages
        for project in self.active_projects.values():
            days_in_stage = (datetime.now() - project.start_date).days
            
            stage_thresholds = {
                ProjectStage.RESEARCH: 90,
                ProjectStage.PROOF_OF_CONCEPT: 60,
                ProjectStage.PROTOTYPE: 90,
                ProjectStage.ALPHA: 60,
                ProjectStage.BETA: 45
            }
            
            threshold = stage_thresholds.get(project.stage, 180)
            
            if days_in_stage > threshold:
                bottlenecks.append({
                    'project_id': project.project_id,
                    'project_name': project.name,
                    'stage': project.stage.value,
                    'days_in_stage': days_in_stage,
                    'threshold': threshold,
                    'recommendation': self._get_bottleneck_recommendation(project)
                })
        
        return bottlenecks
    
    def _get_bottleneck_recommendation(
        self,
        project: ResearchProject
    ) -> str:
        """Generate recommendation for bottlenecked project"""
        
        if project.team_coherence < 0.60:
            return "Low team coherence - consider team restructuring or mediation"
        elif project.production_readiness < 0.50 and project.stage in [ProjectStage.ALPHA, ProjectStage.BETA]:
            return "Low production readiness - focus on code quality, docs, tests"
        elif project.completion_percentage < 0.30:
            return "Low completion - reassess scope or allocate more resources"
        else:
            return "Schedule review meeting with lead researcher and Helena Vu"


# Initialize global pipeline
rd_pipeline = RDProductizationPipeline()
